      const el = origCreate.call(document, tag);
      if (tag === 'canvas') {

        });
        el.toDataURL = () => 'data:image/png;base64,ZZZ';

    const spy = vi.spyOn(exportUtils, 'exportPermitAreaSiteplanV2');

    spy.mockRestore();
  });
